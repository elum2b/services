func TestAuditProbeCloseDuringRun(t *testing.T){
 e:=setup(t,Options{});ctx,cancel:=context.WithTimeout(context.Background(),10*time.Second);defer cancel()
 tx,err:=e.db.BeginTx(ctx,nil);must(t,err);defer tx.Rollback();_,err=tx.ExecContext(ctx,"LOCK TABLE importexport_job IN ACCESS EXCLUSIVE MODE");must(t,err)
 shell:=New();params:=auditRunParams(t,e);done:=make(chan error,1)
 go func(){done<-shell.Run(ctx,params)}()
 defer func(){cancel();select{case <-done:case <-time.After(2*time.Second):t.Error("Run cleanup timeout")}}()
 for{
  var waiting int;must(t,e.db.QueryRowContext(ctx,"SELECT count(*) FROM pg_stat_activity WHERE datname=current_database() AND wait_event_type='Lock' AND query LIKE '%importexport_job%'").Scan(&waiting))
  if waiting>0{break};select{case <-ctx.Done():t.Fatal("Run did not block in archive bootstrap");case <-time.After(5*time.Millisecond):}
 }
 must(t,shell.Close());must(t,tx.Commit())
 deadline:=time.Now().Add(2*time.Second)
 for !shell.IsReady()&&time.Now().Before(deadline){time.Sleep(5*time.Millisecond)}
 if !shell.IsReady(){t.Fatal("defect no longer present: service did not revive")}
 t.Log("CONFIRMED: Close returned, then pending Run adopted a live root and became ready")
}
